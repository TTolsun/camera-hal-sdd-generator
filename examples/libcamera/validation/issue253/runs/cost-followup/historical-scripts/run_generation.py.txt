"""Execute the fixed supplementary cost stages once, retaining failed stages."""
import datetime,hashlib,json,pathlib,shutil,subprocess,sys,time
import yaml
from sdd.config import load

root=pathlib.Path(__file__).resolve().parents[3];work=root/'build/issue253/cost-followup'
runroot=work/'generation-run-01';record=work/'generation-execution-01.json'
if runroot.exists() or record.exists():raise SystemExit('Use a fresh run; existing evidence is never overwritten.')
locked=json.loads((work/'generation-input-lock.json').read_text(encoding='utf-8'))
digest=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
assert all(digest(root/p)==h for p,h in locked['inputs'].items()),'Generator/config/style inputs changed'
assert digest(pathlib.Path(locked['cli_executable']))==locked['cli_sha256']
factsroot=work/'linux-run-02'
for label in ['base','head']:
    model=json.loads((factsroot/(label+'-facts.json')).read_text(encoding='utf-8'))
    assert model['meta']['source_commit']==locked['source_'+label]
runroot.mkdir();rows=[];ledger={'schema':1,'kind':'supplementary-real-update-stages','status':'running','input_lock_sha256':digest(work/'generation-input-lock.json'),'stages':rows,'human_active_review_seconds':None}
def save():record.write_text(json.dumps(ledger,ensure_ascii=False,indent=2),encoding='utf-8')
def stage(name,args,timeout=1800):
    row={'stage':name,'argv':args,'started_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'status':'running','human_active_review_seconds':None,'wait_seconds':None}
    rows.append(row);save();tick=time.perf_counter();print('START '+name,flush=True)
    try:
        with (runroot/(name+'.log')).open('w',encoding='utf-8') as stream:
            p=subprocess.run(args,cwd=root,stdout=stream,stderr=subprocess.STDOUT,timeout=timeout)
        row.update(returncode=p.returncode,status='completed' if not p.returncode else 'failed')
    except Exception as exc:row.update(status='failed',error=repr(exc))
    finally:
        row.update(elapsed_seconds=time.perf_counter()-tick,ended_at=datetime.datetime.now(datetime.timezone.utc).isoformat());save();print('END '+name+' '+json.dumps(row),flush=True)
    if row['status']!='completed':raise RuntimeError(name+' failed; original outputs and logs remain')
def cli(cfg,command,*args):return [sys.executable,str(root/'tools/run_sdd.py'),'--config',str(cfg),command,*args]
def setup(label,previous=None):
    target=runroot/label;target.mkdir();sdd=target/'sdd'
    if previous is None:
        sdd.mkdir()
        for pattern in ['*.md','*.mmd']:
            for p in (work/'document-baseline/docs').rglob(pattern):
                dest=sdd/p.relative_to(work/'document-baseline/docs');dest.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(p,dest)
        shutil.copyfile(work/'catalog-inputs/0',target/'english.json')
    else:
        shutil.copytree(previous/'sdd',sdd);shutil.copyfile(previous/'english.json',target/'english.json')
        if (previous/'build').exists():shutil.copytree(previous/'build',target/'build')
    (target/'facts').mkdir();shutil.copyfile(factsroot/(label+'-facts.json'),target/'facts/facts.json')
    cfg=yaml.safe_load((root/'examples/libcamera/sdd.yaml').read_text(encoding='utf-8'))
    cfg['source']['root']=str(work/'base-source' if label=='base' else root/'build/issue253/source')
    cfg['source']['compile_commands']=str(factsroot/(label+'-compile_commands.json'))
    cfg['paths']={k:str((root/'examples/libcamera'/v).resolve()) for k,v in cfg['paths'].items()}
    cfg['paths']['products_file']=str(root/'examples/libcamera/config/products.yaml')
    cfg['output']={'facts_dir':str(target/'facts'),'sdd_dir':str(sdd),'diagrams_dir':str(sdd/'diagrams')}
    cfg['visuals']['file']=str(root/'examples/libcamera/config/visuals.yaml')
    cfg['site'].update(enabled=False,intro=str(root/'examples/libcamera/site-intro.md'),guide_visuals=str(root/'examples/libcamera/site-guide-visuals.json'),guide_source={'root':str(root),'file':'examples/libcamera/site-intro.md'},english_catalog=str(target/'english.json'),mermaid_dir=str(work/'document-baseline/docs/assets/mermaid'))
    for name,enabled in [('generate',False),('render',True),('review',True)]:
        cfg['site']['enabled']=enabled
        if name=='review':cfg['agent']=yaml.safe_load((root/'docs/libcamera/records/issue253/review.yaml').read_text(encoding='utf-8'))['agent']
        (target/(name+'.yaml')).write_text(yaml.safe_dump(cfg,allow_unicode=True,sort_keys=False),encoding='utf-8')
    return target
try:
    b=setup('base');bcfg=b/'generate.yaml'
    stage('base-reverse-impact',cli(bcfg,'impact','--base',locked['source_head'],'--head',locked['source_base'],'--base-facts',str(factsroot/'head-facts.json'),'--fail-on-coverage-gap'))
    deterministic=[s['id'] for s in load(bcfg).sections() if s.get('kind') in ['index','table','per-package']]
    stage('base-structural-construction',cli(bcfg,'generate','--sections',','.join(deterministic)))
    stage('base-generate-and-carry',cli(bcfg,'generate','--from-impact'))
    stage('base-translate',[sys.executable,'-m','sdd.localization_cli','--config',str(b/'render.yaml'),'--out',str(b/'english.json'),'--translate'])
    stage('base-contract-review',cli(b/'review.yaml','audit-consistency'))
    stage('base-site',cli(b/'render.yaml','export-site','--out',str(b/'site')))
    stage('base-site-verify',cli(b/'render.yaml','verify-site','--out',str(b/'site')))
    h=setup('head',b);hcfg=h/'generate.yaml'
    stage('head-impact',cli(hcfg,'impact','--base',locked['source_base'],'--head',locked['source_head'],'--base-facts',str(b/'facts/facts.json'),'--fail-on-coverage-gap'))
    stage('head-generate-and-carry',cli(hcfg,'generate','--from-impact'))
    stage('head-translate',[sys.executable,'-m','sdd.localization_cli','--config',str(h/'render.yaml'),'--out',str(h/'english.json'),'--translate'])
    stage('head-contract-review',cli(h/'review.yaml','audit-consistency'))
    stage('head-site',cli(h/'render.yaml','export-site','--out',str(h/'site')))
    stage('head-site-verify',cli(h/'render.yaml','verify-site','--out',str(h/'site')))
    first={p.relative_to(h/'site').as_posix():digest(p) for p in (h/'site').rglob('*') if p.is_file()}
    stage('head-site-repeat',cli(h/'render.yaml','export-site','--out',str(h/'site')))
    after={p.relative_to(h/'site').as_posix():digest(p) for p in (h/'site').rglob('*') if p.is_file()}
    ledger['repeat_site_equal']=first==after;assert first==after
    ledger['status']='completed'
except Exception as exc:ledger.update(status='failed',error=repr(exc))
finally:save()
print(json.dumps({'status':ledger['status'],'record':str(record)}),flush=True)
sys.exit(0 if ledger['status']=='completed' else 1)
