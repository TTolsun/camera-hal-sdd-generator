def save():
    raw=json.dumps(ledger,ensure_ascii=False,indent=2)+'\n'
    (export/'execution.json').write_text(raw,encoding='utf-8')
    (native/'execution.json').write_text(raw,encoding='utf-8')
def run(name,argv,cwd=native,timeout=1800):
    started=datetime.datetime.now(datetime.timezone.utc).isoformat();tick=time.perf_counter()
    row={'stage':name,'argv':argv,'cwd':str(cwd),'started_at':started,'status':'running','human_active_review_seconds':None,'wait_seconds':None}
    ledger['stages'].append(row);save();print('START '+name,flush=True)
    target=export/(name+'.log')
    try:
        with target.open('w',encoding='utf-8') as out:
            p=subprocess.run(argv,cwd=cwd,env=env,stdout=out,stderr=subprocess.STDOUT,timeout=timeout)
        row.update(returncode=p.returncode,status='completed' if p.returncode==0 else 'failed')
    except Exception as exc:
        row.update(status='failed',error=repr(exc))
    finally:
        row.update(ended_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),elapsed_seconds=time.perf_counter()-tick)
        save();print('END '+name+' '+json.dumps(row),flush=True)
    if row['status']!='completed': raise RuntimeError('Stage failed: '+name)
def configure(label):
    target=native/label;target.mkdir()
    cfg=yaml.safe_load((generator/'examples/libcamera/sdd.yaml').read_text(encoding='utf-8'))
    cfg['source']['root']=str(source)
    cfg['source']['compile_commands']=str(target/'compile_commands.json')
    cfg['paths']={k:str((generator/'examples/libcamera'/v).resolve()) for k,v in cfg['paths'].items()}
    cfg['paths']['products_file']=str(generator/'examples/libcamera/config/products.yaml')
    cfg['output']={'facts_dir':str(target/'facts'),'sdd_dir':str(target/'sdd'),'diagrams_dir':str(target/'diagrams')}
    path=target/'sdd.yaml';path.write_text(yaml.safe_dump(cfg,allow_unicode=True,sort_keys=False),encoding='utf-8')
    return path
def extract(label,cfg):
    run(label+'-compdb',[sys.executable,str(generator/'examples/libcamera/prepare_compdb.py'),'--build-dir',str(source/'build'),'--out',str(cfg.parent/'compile_commands.json')])

    facts=cfg.parent/'facts/facts.json';data=json.loads(facts.read_text(encoding='utf-8'))
    assert data['meta']['source_commit']==(base if label=='base' else head)
    shutil.copyfile(facts,export/(label+'-facts.json'))
    shutil.copyfile(cfg.parent/'compile_commands.json',export/(label+'-compile_commands.json'))
    shutil.copyfile(cfg,export/(label+'-sdd.yaml'))
    ledger[label+'_facts']={'sha256':hashlib.sha256(facts.read_bytes()).hexdigest(),'meta':{k:v for k,v in data['meta'].items() if k not in ['compile_units','parsed_files']},'qcam_parsed_files':[p for p in data['meta']['parsed_files'] if '/qcam/' in p]}
    save()
