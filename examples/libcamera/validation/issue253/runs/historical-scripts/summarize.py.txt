import hashlib,json,statistics,shutil
from pathlib import Path
root=next(p for p in Path(__file__).resolve().parents if (p/'pyproject.toml').is_file()); r=root/'docs/libcamera/records/issue253'
grades=json.loads((r/'grades.json').read_text(encoding='utf-8'))['grades']
rows=[]
for eid in ['E01','E02','E03']:
    for x in json.loads((r/f'automated/{eid}.json').read_text(encoding='utf-8')):
        g=next(g for g in grades if (g['evaluator_id'],g['task_id'],g['condition'])==(eid,x['task_id'],x['condition']))
        rows.append({**x,'evaluator_id':eid,'grade':g})
summary={'schema':1,'status':'preliminary-automatic-results-human-effect-unmeasured','trials':len(rows),'humans':0,'human_trials':0,'reading_ease_median':None,'effect_verdict':'judgment-unavailable','conditions':{}}
for c in 'AB':
    xs=[x for x in rows if x['condition']==c]
    summary['conditions'][c]={'correct':sum(x['grade']['correct'] for x in xs),'denominator':len(xs),'common_correct':sum(x['grade']['correct'] for x in xs if x['task_id'].startswith('T')),'common_denominator':6,'additional_correct':sum(x['grade']['correct'] for x in xs if x['task_id'].startswith('D')),'additional_denominator':6,'elapsed_median_all_seconds':statistics.median(x['elapsed_seconds'] for x in xs),'help_count':sum(x['help_count'] for x in xs),'first_exposure_correct':sum(x['grade']['correct'] for x in xs if x['exposure']==1),'first_exposure_denominator':sum(x['exposure']==1 for x in xs),'failed_trials':[x['evaluator_id']+'/'+x['task_id'] for x in xs if not x['grade']['correct']]}
pairs=[]
for eid,tid in sorted({(x['evaluator_id'],x['task_id']) for x in rows}):
    a=next(x for x in rows if (x['evaluator_id'],x['task_id'],x['condition'])==(eid,tid,'A'))
    b=next(x for x in rows if (x['evaluator_id'],x['task_id'],x['condition'])==(eid,tid,'B'))
    if a['grade']['correct'] and b['grade']['correct']:pairs.append({'evaluator_id':eid,'task_id':tid,'A':a['elapsed_seconds'],'B':b['elapsed_seconds']})
summary['both_correct_pairs']=pairs
summary['both_correct_medians']={c:statistics.median(x[c] for x in pairs) for c in 'AB'}
summary['limits']=['Automatic grades are provisional: precise attribution to every cited HTML and device/test source was not independently re-opened. See grades.json limitations.','Three model roles are not three human participants or independent model families.','Second exposure is contaminated by task learning; first exposure is not a balanced difficulty/role sample.','B index start is not documented in three E01 trials. E02 shell-reported wall times are not a pure environment-delay measure.','No new post-fix trial or human reading-ease measurement occurred.']
(r/'summary.json').write_text(json.dumps(summary,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps(summary,ensure_ascii=False))
